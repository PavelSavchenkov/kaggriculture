cmake_minimum_required(VERSION 3.18)
project(late_animal_schedule LANGUAGES CXX)
get_filename_component(EXPERIMENT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(REPOSITORY "${EXPERIMENT}/../../.." ABSOLUTE)
add_subdirectory("${REPOSITORY}/day_solver" day_solver)
add_executable(retry retry.cpp)
target_include_directories(retry BEFORE PRIVATE "${REPOSITORY}")
target_compile_options(retry PRIVATE -O3 -march=native -mtune=native)
target_link_libraries(retry PRIVATE DaySolver::scheduler)
foreach(leaf off on)
    add_executable(audit_${leaf} audit.cpp
        "${EXPERIMENT}/league/top_replay_library/source/agent.cpp"
        "${EXPERIMENT}/league/public_router/source/agent.cpp")
    target_include_directories(audit_${leaf} BEFORE PRIVATE "${REPOSITORY}")
    target_compile_options(audit_${leaf} PRIVATE -O3 -march=native -mtune=native)
    target_link_libraries(audit_${leaf} PRIVATE DaySolver::scheduler)
endforeach()
target_compile_definitions(audit_on PRIVATE LATE_BERRY=1)

add_executable(reuse reuse.cpp)
target_include_directories(reuse BEFORE PRIVATE "${REPOSITORY}")
target_compile_options(reuse PRIVATE -O3 -march=native -mtune=native)
target_link_libraries(reuse PRIVATE DaySolver::scheduler)
add_executable(retime retime.cpp)
target_include_directories(retime BEFORE PRIVATE "${REPOSITORY}")
target_compile_options(retime PRIVATE -O3 -march=native -mtune=native)
target_link_libraries(retime PRIVATE DaySolver::scheduler)
