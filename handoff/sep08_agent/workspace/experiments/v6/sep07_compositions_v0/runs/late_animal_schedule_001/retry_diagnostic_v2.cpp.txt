#include "day_solver/scheduler.hpp"
#include "day_solver/io.hpp"
#include <fstream>
#include <iostream>
#include <iomanip>

using namespace kag;
using namespace day_solver;
namespace fs=std::filesystem;

std::array<Action,24> read_actions(const fs::path& path) {
    std::ifstream input(path);if(!input)std::abort();
    std::array<Action,24> actions;
    for(auto& a:actions) {
        input>>a.n_units>>a.n_orders;
        for(int u=0;u<a.n_units;++u) {
            int op,arg;input>>op>>arg>>a.units[u].n;a.units[u].op=op;a.units[u].arg=arg;
        }
        for(int s=0;s<a.n_orders;++s) {
            int op,item;input>>op>>item>>a.orders[s].n;a.orders[s].op=op;a.orders[s].item=item;
        }
        a.finalize();
    }
    if(!input)std::abort();return actions;
}

void write_actions(const fs::path& path,const std::array<Action,24>& actions) {
    std::ofstream out(path);
    for(const auto& a:actions) {
        out<<a.n_units<<' '<<a.n_orders;
        for(int u=0;u<a.n_units;++u)out<<' '<<+a.units[u].op<<' '<<+a.units[u].arg<<' '<<a.units[u].n;
        for(int s=0;s<a.n_orders;++s)out<<' '<<+a.orders[s].op<<' '<<+a.orders[s].item<<' '<<a.orders[s].n;
        out<<'\n';
    }
}

int main(int argc,char** argv) {
    if(argc!=5 || fs::exists(argv[3]))return 2;
    const auto problem=load_problem_json(argv[1]);
    const auto original=read_actions(argv[2]);
    const fs::path out(argv[3]);fs::create_directories(out);
    day_scheduler::Options options;options.seconds=std::stod(argv[4]);options.fallback_workers=4;
    const auto result=day_scheduler::solve(problem,options);
    std::ofstream status(out/"STATUS.json");
    status<<"{\"worker_count\":"<<problem.worker_count<<",\"budget_seconds\":"<<options.seconds
          <<",\"elapsed_seconds\":"<<result.seconds<<",\"physical_schedule\":"<<bool(result.schedule);
    if(!result.schedule){status<<"}\n";std::cout<<"UNKNOWN "<<result.seconds<<'\n';return 0;}
    auto actions=*result.schedule;
    for(int h=0;h<24;++h) {
        actions[h].n_orders=original[h].n_orders;
        std::copy_n(original[h].orders,original[h].n_orders,actions[h].orders);
        for(int s=0;s<actions[h].n_orders;++s)if(actions[h].orders[s].op==M_HIRE)actions[h].orders[s]={};
    }
    for(const auto& order:problem.market_plan)if(order.market_op==M_HIRE) {
        auto& action=actions[order.hour];
        action.orders[order.order_index]={M_HIRE,0,order.quantity};
        action.n_orders=std::max<int>(action.n_orders,order.order_index+1);
    }
    for(auto& action:actions)action.finalize();
    const auto replay=replay_schedule(problem,actions);
    status<<",\"requirements\":"<<replay.requirements_satisfied<<",\"invariants\":"<<replay.invariants_satisfied<<"}\n";
    for(const auto& error:replay.errors)std::cerr<<error<<"\n";
    if(!replay.requirements_satisfied || !replay.invariants_satisfied){write_actions(out/"invalid_actions.txt",actions);return 3;}
    write_actions(out/"actions.txt",actions);
    std::cout<<"SCHEDULE "<<result.seconds<<'\n';
}
