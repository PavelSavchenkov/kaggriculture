#include "experiments/v6/sep07_compositions_v0/include/biology.hpp"
#include <fstream>
#include <iostream>

using namespace kag;
using namespace compositions;

Biology actual(int first,const Service& service){
    Config config;config.seed=42;Sim sim(config);sim.st.day=first;sim.st.step=first*24;
    auto& farm=sim.st.farms[0];farm.shed[COW]=1;farm.shed_total=1;farm.inv_add(0,WHEAT,50);
    Biology result;std::array<int,N_PRODUCTS> previous{};
    while(!sim.st.done){
        const int day=sim.st.day,hour=sim.st.hour;Action own,other;
        own.n_units=other.n_units=1;
        const int offset=day==first?3:0;
        if(day==first && hour==0)own.units[0]={OP_BUILD_PASTURE,0,1};
        if(day==first && hour==1)own.units[0]={OP_PICKUP,COW,1};
        if(day==first && hour==2)own.units[0]={OP_PLACE,COW,1};
        if(hour==offset && on(service.feed,day))own.units[0]={OP_FEED,0,1};
        if(hour==offset+1 && on(service.care,day))own.units[0]={OP_CARE,0,1};
        if(hour==offset+2 && day>first)own.units[0]={OP_COLLECT_FERTILIZER,0,1};
        if(hour==offset+3 && day>first)own.units[0]={OP_HARVEST,0,1};
        own.finalize();other.finalize();sim.step(own,other);
        if(sim.st.day!=day || sim.st.done){
            for(int i=0;i<N_PRODUCTS;++i){result.days[day].output[i]=farm.produced[i]-previous[i];previous[i]=farm.produced[i];}
        }
    }
    return result;
}

int main(int argc,char** argv){
    if(argc!=2)return 2;std::ofstream out(argv[1]);
    out<<"birth,mode,engine_matches_biology,milk,fertilizer,operations\n";
    for(int birth=8;birth<=21;++birth){
        Service original;original.feed&=~(1u<<29);
        int last=birth+8;while(last+2<=29)last+=2;
        original.care&=(1u<<(last-1))-1;
        Service revised=original;revised.care&=~((1u<<(birth+1))|(1u<<(birth+2)));
        const auto a=biology({COW,1,birth,30},original);
        const auto b=biology({COW,1,birth,30},revised);
        for(int mode=0;mode<2;++mode){
            const auto predicted=mode?b:a;const auto engine=actual(birth,mode?revised:original);
            int milk=0,fert=0,ops=0;
            for(int d=birth;d<30;++d){
                for(int i=0;i<N_PRODUCTS;++i)
                    if(engine.days[d].output[i]!=predicted.days[d].output[i] || a.days[d].output[i]!=b.days[d].output[i]){std::cerr<<"birth "<<birth<<" mode "<<mode<<" day "<<d<<" item "<<i<<" engine "<<engine.days[d].output[i]<<" predicted "<<predicted.days[d].output[i]<<" original "<<a.days[d].output[i]<<" revised "<<b.days[d].output[i]<<std::endl;std::abort();}
                milk+=predicted.days[d].output[MILK];fert+=predicted.days[d].output[FERTILIZER];ops+=predicted.days[d].operations;
            }
            out<<birth<<','<<mode<<",1,"<<milk<<','<<fert<<','<<ops<<'\n';
        }
    }
    std::cout<<"28 engine/model checks;14 care equivalences; two actions saved per cow."<<std::endl;
}
