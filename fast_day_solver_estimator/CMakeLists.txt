cmake_minimum_required(VERSION 3.19)
project(fast_day_solver_estimator LANGUAGES CXX)

set(FAST_ESTIMATOR_DAY_SOLVER "${CMAKE_CURRENT_SOURCE_DIR}/../day_solver" CACHE PATH "Shared repository solver/engine dependency")
set(FAST_ESTIMATOR_AGENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Root containing the published agents directory")
option(FAST_ESTIMATOR_BUILD_WARM "Build full-course benchmark fixtures" OFF)
add_subdirectory("${FAST_ESTIMATOR_DAY_SOLVER}" day_solver EXCLUDE_FROM_ALL)

add_library(fast_day_estimator INTERFACE)
add_library(FastDayEstimator::estimator ALIAS fast_day_estimator)
target_compile_features(fast_day_estimator INTERFACE cxx_std_20)
target_include_directories(fast_day_estimator INTERFACE
  "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${CMAKE_CURRENT_SOURCE_DIR}/models/search_candidate_dev_v2"
  "${CMAKE_CURRENT_SOURCE_DIR}/models/context_candidate_dev_v2"
  "${FAST_ESTIMATOR_DAY_SOLVER}/include"
  "${FAST_ESTIMATOR_DAY_SOLVER}/src/components/resource_portfolio_order_v2/solver/include")

function(estimator_target name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE FastDayEstimator::estimator)
  target_compile_options(${name} PRIVATE -O3 -Wall -Wextra -Wpedantic)
endfunction()

estimator_target(estimate_day source/estimate_day.cpp)
target_link_libraries(estimate_day PRIVATE DaySolver::scheduler)
estimator_target(estimator_api_smoke tests/api_smoke.cpp)
enable_testing()
add_test(NAME estimator_api COMMAND estimator_api_smoke)

foreach(name predict_planning predict_context predict_search reference reference_terminal_deadlines replay_search_cpp replay_refinement_cpp replay_long_retry_cpp check_planning check_context_query check_terminal_deadlines check_seed_suffix)
  add_executable(${name} EXCLUDE_FROM_ALL source/${name}.cpp)
  target_link_libraries(${name} PRIVATE FastDayEstimator::estimator DaySolver::scheduler)
  target_compile_options(${name} PRIVATE -O3 -Wall -Wextra -Wpedantic)
endforeach()
target_compile_definitions(reference_terminal_deadlines PRIVATE LABOR_ACTIVE_HOURS=23)

if(FAST_ESTIMATOR_BUILD_WARM)
  set(agent_sources)
  foreach(name early_structure_cow nanare_four_quadrant_course)
    set(directory "${FAST_ESTIMATOR_AGENT_ROOT}/agents/external/${name}")
    file(READ "${directory}/agent.json" manifest)
    string(JSON count LENGTH "${manifest}" sources)
    math(EXPR last "${count} - 1")
    foreach(index RANGE ${last})
      string(JSON path GET "${manifest}" sources ${index})
      list(APPEND agent_sources "${directory}/${path}")
    endforeach()
  endforeach()
  add_library(fixture_agents OBJECT ${agent_sources})
  target_link_libraries(fixture_agents PRIVATE FastDayEstimator::estimator)
  target_include_directories(fixture_agents PRIVATE "${FAST_ESTIMATOR_AGENT_ROOT}")
  target_compile_options(fixture_agents PRIVATE -O3)
  foreach(name fixture_generic fixture_pair)
    estimator_target(${name} tests/fixture_agents.cpp)
    target_sources(${name} PRIVATE $<TARGET_OBJECTS:fixture_agents>)
    target_include_directories(${name} PRIVATE "${FAST_ESTIMATOR_AGENT_ROOT}")
    target_compile_options(${name} PRIVATE -UNDEBUG)
  endforeach()
  target_compile_definitions(fixture_pair PRIVATE FAST_ESTIMATOR_TYPED_FIXTURE)
  foreach(name warm_season warm_shop_season verify_warm_course verify_warm_shop_course warm_compile_early warm_compile_guided warm_compile_stage warm_shop_compile_original warm_shop_compile_guided warm_shop_compile_stage)
    estimator_target(${name} source/${name}.cpp)
    target_sources(${name} PRIVATE $<TARGET_OBJECTS:fixture_agents>)
    target_include_directories(${name} BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${FAST_ESTIMATOR_AGENT_ROOT}")
    target_include_directories(${name} SYSTEM PRIVATE "${FAST_ESTIMATOR_DAY_SOLVER}/vendor/include")
    target_compile_definitions(${name} PRIVATE OR_PROTO_DLL= PROTOBUF_USE_DLLS)
    target_link_libraries(${name} PRIVATE DaySolver::scheduler)
  endforeach()
endif()
