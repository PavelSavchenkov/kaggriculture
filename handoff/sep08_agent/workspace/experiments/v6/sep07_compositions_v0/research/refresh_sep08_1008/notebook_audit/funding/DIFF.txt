+    def _current_purchase_cost(self, action, view):
+        cost, hires, parcels = 0.0, 0, 0
+        for order in action.get("market", []):
+            if not order:

+            op = order[0]

+                    cost += LAND_PRICES[index]

+                    cost += ANIMAL_COST.get(item, 0) * qty

+                    cost += view.prices.get(item, 0) * qty
+        return cost
