cmake_minimum_required(VERSION 3.18)
project(day_policy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(DAY_POLICY_NATIVE "Tune Release for the build CPU" ON)
option(DAY_POLICY_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(DAY_POLICY_SANITIZE)
  add_compile_options(-O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(CMAKE_BUILD_TYPE STREQUAL "Release")
  add_compile_options(-O3 -fno-math-errno -fno-semantic-interposition -fno-plt -flto)
  add_link_options(-flto)
  if(DAY_POLICY_NATIVE)
    add_compile_options(-march=native -mtune=native)
  endif()
endif()

add_library(contract_policy STATIC source/agent.cpp source/routes.cpp source/policy.cpp source/verify.cpp)
target_include_directories(contract_policy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/vendor" "${CMAKE_CURRENT_SOURCE_DIR}/source")
target_compile_options(contract_policy PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
add_library(day_policy::policy ALIAS contract_policy)

include(CTest)
if(BUILD_TESTING)
  add_executable(contract_regressions tests/regressions.cpp)
  target_link_libraries(contract_regressions PRIVATE contract_policy)
  add_test(NAME contract_regressions COMMAND contract_regressions)
  foreach(tool IN ITEMS extract evaluate progression return_bounds placement_patterns placement_cost)
    add_executable(contract_${tool} tools/${tool}.cpp)
    target_link_libraries(contract_${tool} PRIVATE contract_policy)
  endforeach()
  find_package(Threads REQUIRED)
  add_executable(contract_arena tools/arena.cpp)
  target_link_libraries(contract_arena PRIVATE contract_policy Threads::Threads)
  add_executable(contract_pair tools/arena.cpp)
  target_compile_definitions(contract_pair PRIVATE PAIR_RUNNER=1)
  target_link_libraries(contract_pair PRIVATE contract_policy Threads::Threads)
endif()
