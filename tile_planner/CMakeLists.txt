cmake_minimum_required(VERSION 3.19)
project(tile_planner LANGUAGES CXX)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_subdirectory("${REPO}/fast_day_solver_estimator" estimator EXCLUDE_FROM_ALL)

add_library(tile_planner INTERFACE)
add_library(TilePlanner::planner ALIAS tile_planner)
target_compile_features(tile_planner INTERFACE cxx_std_20)
target_include_directories(tile_planner INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${REPO}")
target_link_libraries(tile_planner INTERFACE FastDayEstimator::estimator DaySolver::scheduler)

set(core_tools compile_cold export_calendar import_lives search_lives probe_lives solve_point pack_bank)
file(GLOB tool_sources CONFIGURE_DEPENDS "source/*.cpp")
foreach(source IN LISTS tool_sources)
  get_filename_component(name "${source}" NAME_WE)
  if(name IN_LIST core_tools)
    add_executable(${name} "${source}")
  else()
    add_executable(${name} EXCLUDE_FROM_ALL "${source}")
  endif()
  target_link_libraries(${name} PRIVATE TilePlanner::planner)
  target_compile_options(${name} PRIVATE -O3 -Wall -Wextra -Wpedantic -fno-math-errno)
endforeach()
target_sources(compile_cold PRIVATE "${REPO}/agents/external/public_router/source/deps/league/public_router/source/agent.cpp")
