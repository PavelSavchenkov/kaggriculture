#include "season.hpp"
#include "experiments/v6/sep07_compositions_v0/include/animal_investment_value.hpp"

std::vector<TileWorkAction> work_at(const DayProblem& problem, int cell) {
    for (const auto& work : problem.tile_work) if (work.tile == cell) return work.actions;
    return {};
}

int main(int argc, char** argv) {
    if (argc != 3 || fs::exists(argv[1])) return 2;
    const fs::path out(argv[1]); fs::create_directories(out);
    const Season source(std::stoull(argv[2]));
    for (const auto& life : source.lives) if (!life.exact) return 3;
    FarmFlowPlan plan;
    for (int day=0; day<30; ++day) for (const auto& action : source.days[day].own)
        for (int j=0; j<action.n_orders; ++j) {
            const auto& order=action.orders[j];
            if (order.op==M_SELL) plan.sales[day][order.item]+=order.n;
            if (order.op==M_BUY_PRODUCT) plan.buys[day][order.item]+=order.n;
        }
    std::ofstream csv(out / "proposals.csv");
    csv << "cell,entry_day,animal,animal_output,wheat_use,fertilizer_net,field_operations_delta,minimum_daily_travel,source_visit_deficit,observed_own_gain,observed_margin_gain,offline_shadow_gain,estimate_seconds\n";
    int count=0;
    for (int day=13; day<=22; ++day) for (int cell=0; cell<100; ++cell) {
        const auto& start=source.days[day].start;
        const auto& tile=start.st.farms[0].tiles[cell/10][cell%10];
        if (tile.kind!=T_PLANT || CROPS[tile.what].ongoing) continue;
        const auto original=work_at(source.days[day].problem,cell);
        bool harvest=false,replant=false;
        for (const auto& job : original) {
            if (job.op==OP_HARVEST) harvest=true;
            if (job.op==OP_PLANT && harvest) replant=true;
        }
        if (!replant) continue;
        const auto observation=agent::runtime::make_observation(start,0);
        Biology empty;
        const auto control=value_animal_investment(observation,plan,empty);
        for (int item=GOOSE; item<=SHEEP; ++item) {
            const auto begin=std::chrono::steady_clock::now();
            Service service; service.feed &= ~(1u<<29); service.care &= ~(1u<<29);
            const auto& species=ANIMALS[item-GOOSE];
            int last_production=day+species.first_yield_day;
            while (last_production+species.interval<=29) last_production+=species.interval;
            // Care earned after this point cannot affect output before game end.
            service.care &= (1u<<std::max(0,last_production-1))-1;
            auto addition=biology({uint8_t(item),1,day,30},service);
            const int animal_cost=addition.animal_cost;
            int seeds_saved=0, operations=0, output=0, feed=0, fertilizer=0, deficit=0;
            for (const auto& life : source.lives) if (life.cell==cell && life.cohort.start_day>=day) {
                const auto removed=biology(life.cohort,life.service); seeds_saved+=removed.seed_cost;
                for (int d=day; d<30; ++d) {
                    for (int p=0; p<N_PRODUCTS; ++p) addition.days[d].output[p]-=removed.days[d].output[p];
                    addition.days[d].output[FERTILIZER]+=removed.days[d].fertilizer;
                    addition.days[d].operations-=removed.days[d].operations;
                }
            }
            double shadow=-animal_cost+seeds_saved;
            for (int d=day; d<30; ++d) {
                const auto& delta=addition.days[d];
                output+=delta.output[species.product]; feed+=delta.wheat;
                fertilizer+=delta.output[FERTILIZER]; operations+=delta.operations;
                const auto& inv=source.days[d].start.st.market.inventory;
                shadow-=delta.wheat*market_price(WHEAT,inv[WHEAT]);
                for (int p=0; p<N_PRODUCTS; ++p) shadow+=delta.output[p]*market_price(p,inv[p]);
                auto original_work=work_at(source.days[d].problem,cell);
                int visits=0; auto replay=source.days[d].start;
                for (int h=0; h<24 && !replay.st.done; ++h) {
                    const auto& farm=replay.st.farms[0];
                    for (int u=0; u<farm.n_units; ++u) {
                        const auto op=source.days[d].own[h].units[u].op;
                        if (farm.pos_x[u]==cell%10 && farm.pos_y[u]==cell/10 &&
                            (op==OP_PASS || (op>=OP_PLANT && op<=OP_CARE))) ++visits;
                    }
                    replay.step(source.days[d].own[h],source.days[d].rival[h]);
                }
                deficit+=std::max(0,int(original_work.size())+delta.operations-visits);
            }
            const auto value=value_animal_investment(observation,plan,addition);
            const int x=cell%10,y=cell/10;
            const int distance=std::min(std::abs(x-3),std::abs(x-4))+std::min(std::abs(y-3),std::abs(y-4));
            csv << cell << ',' << day << ',' << item << ',' << output << ',' << feed << ',' << fertilizer << ',' << operations << ','
                << 2*distance << ',' << deficit << ',' << value.own-control.own+seeds_saved << ','
                << value.margin()-control.margin()+seeds_saved << ',' << shadow << ','
                << std::chrono::duration<double>(std::chrono::steady_clock::now()-begin).count() << '\n';
            ++count;
        }
    }
    std::ofstream(out / "STATUS.json") << "{\"source_crop_lives_exact\":" << source.lives.size() << ",\"proposals\":" << count << "}\n";
    std::cout << "proposals=" << count << " source lives=" << source.lives.size() << '\n';
}
